المعادلة الحسابية لتغطية الشاشة ومنع الفراغات (Full-Bleed Placement Solver):

// Dynamic scale & placement equation in public/index.html
const h = Math.max(
  vh,
  vw / art.aspect,
  diameter / art.d,
  ay / art.cy,
  (vh - ay) / (1 - art.cy),
  (ax / art.cx) / art.aspect,
  ((vw - ax) / (1 - art.cx)) / art.aspect
);
const w = h * art.aspect;
const left = ax - art.cx * w;
const top = ay - art.cy * h;

5. مستودع GitHub والتشغيل 

# 1. Clone the repository
git clone https://github.com/RadioBahrain/player.git
cd player

# 2. Install dependencies & run development server
bun install
bun run dev

# 3. Run automated visual E2E test suite (50 assertions)
bun test

# 4. Generate high-resolution retina screenshot gallery
bun run screenshots

# 5. Build and deploy to production
bun run deploy:live
